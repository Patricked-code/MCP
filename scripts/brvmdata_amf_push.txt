#!/usr/bin/env python3
import json, os, shutil, subprocess, sys
from pathlib import Path

MODE = sys.argv[1] if len(sys.argv) > 1 else ""
REPO_SSH = "git@github.com:chainsolutions-wealthtech/BRVMDATA.git"
REPO_HTTPS = "https://github.com/chainsolutions-wealthtech/BRVMDATA.git"
BRANCH = "data/amf-umoa-registre-natif-20260805"
WORK = Path("/tmp/brvmdata-amf-push")
ROOT = Path("/tmp/amf_registry_native")
ARCHIVE = Path("/var/www/vhosts/wealthtechinnovations.com/httpdocs/downloads/amf-umoa/AMF_UMOA_REGISTRE_DYNAMIQUE_NATIF_COMPLET_2026-08-05_6f18c91e7a24.zip")
SQLITE_GZ = Path("/tmp/AMF_UMOA_REGISTRE_DYNAMIQUE_NATIF.sqlite.gz")
DEST = Path("data/amf-umoa/registre-dynamique/2026-08-05")

def run(cmd, **kw):
    return subprocess.run(cmd, text=True, capture_output=True, **kw)

def preflight():
    ssh = run(["git","ls-remote",REPO_SSH,"HEAD"], env={**os.environ,"GIT_SSH_COMMAND":"ssh -o BatchMode=yes -o StrictHostKeyChecking=accept-new -o ConnectTimeout=20"})
    https = run(["git","ls-remote",REPO_HTTPS,"HEAD"], env={**os.environ,"GIT_TERMINAL_PROMPT":"0"})
    lfs = run(["git","lfs","version"])
    out = {
      "ssh_auth_ok": ssh.returncode == 0,
      "https_auth_ok": https.returncode == 0,
      "git_lfs_available": lfs.returncode == 0,
      "git_lfs_version": (lfs.stdout+lfs.stderr)[-500:],
      "sources": [{"path":str(p),"exists":p.exists(),"size_bytes":p.stat().st_size if p.is_file() else None} for p in (ARCHIVE,ROOT,SQLITE_GZ)]
    }
    print(json.dumps(out, indent=2))
    return out

def push():
    pf = preflight()
    if not (pf["ssh_auth_ok"] or pf["https_auth_ok"]): raise SystemExit("GitHub auth unavailable")
    if not all(x["exists"] for x in pf["sources"]): raise SystemExit("AMF source missing")
    repo = REPO_SSH if pf["ssh_auth_ok"] else REPO_HTTPS
    shutil.rmtree(WORK, ignore_errors=True)
    env = os.environ.copy()
    if repo == REPO_SSH: env["GIT_SSH_COMMAND"]="ssh -o BatchMode=yes -o StrictHostKeyChecking=accept-new -o ConnectTimeout=20"
    else: env["GIT_TERMINAL_PROMPT"]="0"
    subprocess.run(["git","clone","--branch",BRANCH,"--single-branch",repo,str(WORK)],check=True,env=env)
    os.chdir(WORK)
    subprocess.run(["git","config","user.name","ChainSolutions Data Import"],check=True)
    subprocess.run(["git","config","user.email","41898282+github-actions[bot]@users.noreply.github.com"],check=True)
    shutil.rmtree(DEST, ignore_errors=True)
    for d in ("archive","database","exports","metadata"): (DEST/d).mkdir(parents=True,exist_ok=True)
    for d in ("csv","jsonl"):
        if (ROOT/d).exists(): shutil.copytree(ROOT/d,DEST/"exports"/d)
    for f in ("README.md","EXTRACTION_METADATA.json","SCHEMA.sql","FILE_MANIFEST_SHA256.csv"):
        if (ROOT/f).exists(): shutil.copy2(ROOT/f,DEST/"metadata"/f)
    if (ROOT/"AMF_UMOA_REGISTRE_DYNAMIQUE_NATIF.sqlite").exists(): shutil.copy2(ROOT/"AMF_UMOA_REGISTRE_DYNAMIQUE_NATIF.sqlite",DEST/"database")
    shutil.copy2(SQLITE_GZ,DEST/"database")
    shutil.copy2(ARCHIVE,DEST/"archive")
    lfs_used = pf["git_lfs_available"]
    split_used = False
    if lfs_used:
        subprocess.run(["git","lfs","install","--local"],check=True)
        subprocess.run(["git","lfs","track","*.zip","*.sqlite","*.gz"],check=True)
    else:
        target=DEST/"archive"/ARCHIVE.name
        subprocess.run(["split","-b","90M","-d","-a","3",str(target),str(target)+".part-"],check=True)
        target.unlink()
        split_used=True
    files=[p for p in DEST.rglob("*") if p.is_file()]
    with (DEST/"SHA256SUMS.txt").open("w") as o:
        for p in sorted(files):
            h=run(["sha256sum",str(p)]).stdout.split()[0]
            o.write(f"{h}  {p.relative_to(DEST)}\n")
    (DEST/"README.md").write_text(f"""# Registre dynamique natif AMF-UMOA — 2026-08-05

Extraction native de `https://www.amf-umoa.org/accueil/intervenant`.

- 164 intervenants
- 237 OPC/FIA/FCTC
- 100 commissaires aux comptes
- SQLite integrity_check: ok
- Git LFS: {str(lfs_used).lower()}
- Archive découpée: {str(split_used).lower()}

Contenu : archive complète, SQLite, CSV, JSONL, schéma SQL, métadonnées et checksums.

Si l'archive est découpée :
```bash
cat archive/*.zip.part-* > archive/{ARCHIVE.name}
```
""",encoding="utf-8")
    subprocess.run(["git","add",".gitattributes",str(DEST)],check=False)
    if not (WORK/".gitattributes").exists(): subprocess.run(["git","add",str(DEST)],check=True)
    subprocess.run(["git","commit","-m","data(amf-umoa): add native registry snapshot and archive"],check=True)
    subprocess.run(["git","push","origin",BRANCH],check=True,env=env)
    sha=run(["git","rev-parse","HEAD"]).stdout.strip()
    print(json.dumps({"branch":BRANCH,"commit_sha":sha,"git_lfs_used":lfs_used,"archive_split":split_used,"files":len([p for p in DEST.rglob("*") if p.is_file()])},indent=2))

if MODE=="preflight": preflight()
elif MODE=="push": push()
else: raise SystemExit("usage: preflight|push")
